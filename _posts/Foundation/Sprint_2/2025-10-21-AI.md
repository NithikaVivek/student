#### 3. Error Handling and Fallbacks

```javascript
async function robustAICall(message, retries = 3) {
  for (let i = 0; i < retries; i++) {
    try {
      const response = await aiChat.sendMessage(message);
      return response;
    } catch (error) {
      if (error.status === 429) {
        // Rate limited - wait and retry
        await sleep(Math.pow(2, i) * 1000);
        continue;
      } else if (error.status === 500) {
        // Server error - retry
        continue;
      } else {
        // Client error - don't retry
        throw error;
      }
    }
  }
  
  // All retries failed
  return "I'm having trouble connecting right now. Please try again.";
}
```

## Performance Optimization

### 1. Debouncing User Input

```javascript
function debounce(func, wait) {
  let timeout;
  return function executedFunction(...args) {
    clearTimeout(timeout);
    timeout = setTimeout(() => func.apply(this, args), wait);
  };
}

const debouncedSearch = debounce(async (query) => {
  const suggestions = await aiChat.getSuggestions(query);
  displaySuggestions(suggestions);
}, 300);
```

### 2. Caching Responses

```javascript
class ResponseCache {
  constructor(ttl = 3600000) { // 1 hour default
    this.cache = new Map();
    this.ttl = ttl;
  }
  
  set(key, value) {
    this.cache.set(key, {
      value,
      timestamp: Date.now()
    });
  }
  
  get(key) {
    const item = this.cache.get(key);
    if (!item) return null;
    
    if (Date.now() - item.timestamp > this.ttl) {
      this.cache.delete(key);
      return null;
    }
    
    return item.value;
  }
}
```

### 3. Streaming Responses

```javascript
async function streamResponse(prompt) {
  const response = await fetch('/api/chat/stream', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ prompt })
  });
  
  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  
  let partialMessage = '';
  
  while (true) {
    const { value, done } = await reader.read();
    if (done) break;
    
    const chunk = decoder.decode(value);
    partialMessage += chunk;
    
    // Update UI with partial message
    updateMessage(partialMessage);
  }
  
  return partialMessage;
}
```

### Rate Limiting

```javascript
class RateLimiter {
  constructor(maxRequests, windowMs) {
    this.maxRequests = maxRequests;
    this.windowMs = windowMs;
    this.requests = new Map();
  }
  
  async checkLimit(userId) {
    const now = Date.now();
    const userRequests = this.requests.get(userId) || [];
    
    // Remove old requests outside window
    const validRequests = userRequests.filter(
      time => now - time < this.windowMs
    );
    
    if (validRequests.length >= this.maxRequests) {
      const oldestRequest = Math.min(...validRequests);
      const waitTime = this.windowMs - (now - oldestRequest);
      throw new Error(`Rate limit exceeded. Try again in ${Math.ceil(waitTime / 1000)}s`);
    }
    
    validRequests.push(now);
    this.requests.set(userId, validRequests);
  }
}
```

## Cost Management

### Token Usage Tracking

```javascript
class TokenTracker {
  constructor() {
    this.usage = {
      promptTokens: 0,
      completionTokens: 0,
      totalCost: 0
    };
  }
  
  recordUsage(response) {
    this.usage.promptTokens += response.usage.prompt_tokens;
    this.usage.completionTokens += response.usage.completion_tokens;
    
    // GPT-4 pricing example (adjust for your model)
    const promptCost = (response.usage.prompt_tokens / 1000) * 0.03;
    const completionCost = (response.usage.completion_tokens / 1000) * 0.06;
    
    this.usage.totalCost += promptCost + completionCost;
  }
  
  getReport() {
    return {
      ...this.usage,
      averageCostPerRequest: this.usage.totalCost / this.requests
    };
  }
}
```

## Lessons Learned

### 1. Start Simple
Don't try to build AGI on day one. Start with focused, well-defined use cases.

### 2. Prompt Engineering Matters
Spend time crafting good prompts. A well-designed prompt can make the difference between mediocre and excellent results.

### 3. Handle Failures Gracefully
AI services will fail. Have fallbacks, retries, and clear error messages.

### 4. Monitor Costs
AI API calls add up quickly. Implement caching, rate limiting, and usage tracking.

### 5. User Experience First
The AI should enhance UX, not complicate it. Keep interfaces simple and responsive.
